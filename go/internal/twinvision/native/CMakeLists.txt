cmake_minimum_required(VERSION 3.16)
project(WendyTwinVision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc features2d calib3d flann)
add_library(twin-localization STATIC SiteLocalization.cpp)
target_include_directories(twin-localization PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(twin-localization PUBLIC ${OpenCV_LIBS})
add_executable(wendy-twin-vision main.cpp)
target_link_libraries(wendy-twin-vision PRIVATE twin-localization)
add_executable(test-vision Tests.cpp)
# These are assertions, including in Release builds.
target_compile_options(test-vision PRIVATE -UNDEBUG)
target_link_libraries(test-vision PRIVATE twin-localization)
enable_testing()
add_test(NAME localization COMMAND test-vision)
install(TARGETS wendy-twin-vision RUNTIME DESTINATION bin)
