cmake_minimum_required(VERSION 3.16)
project(world_go2_sensors LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(world-go2-sensors sensors.cpp)
target_compile_features(world-go2-sensors PRIVATE cxx_std_17)
ament_target_dependencies(world-go2-sensors rclcpp nav_msgs sensor_msgs std_msgs tf2_msgs)
install(TARGETS world-go2-sensors RUNTIME DESTINATION bin)
