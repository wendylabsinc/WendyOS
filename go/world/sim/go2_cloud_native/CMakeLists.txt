cmake_minimum_required(VERSION 3.16)
project(world_go2_cloud_observer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(world-go2-cloud-observer cloud.cpp)
target_compile_features(world-go2-cloud-observer PRIVATE cxx_std_17)
ament_target_dependencies(world-go2-cloud-observer rclcpp nav_msgs sensor_msgs std_msgs tf2_msgs)
install(TARGETS world-go2-cloud-observer RUNTIME DESTINATION bin)
add_executable(world-go2-cloud-shadow cloud.cpp)
target_compile_definitions(world-go2-cloud-shadow PRIVATE WORLD_GO2_CLOUD_SHADOW=1)
target_compile_features(world-go2-cloud-shadow PRIVATE cxx_std_17)
ament_target_dependencies(world-go2-cloud-shadow rclcpp nav_msgs sensor_msgs std_msgs tf2_msgs)
install(TARGETS world-go2-cloud-shadow RUNTIME DESTINATION bin)
add_executable(world-go2-cloud-live cloud.cpp)
target_compile_definitions(world-go2-cloud-live PRIVATE WORLD_GO2_CLOUD_LIVE=1)
target_compile_features(world-go2-cloud-live PRIVATE cxx_std_17)
ament_target_dependencies(world-go2-cloud-live rclcpp nav_msgs sensor_msgs std_msgs tf2_msgs)
install(TARGETS world-go2-cloud-live RUNTIME DESTINATION bin)
